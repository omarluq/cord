package main

import (
	"context"
	"database/sql"
	"errors"
	"fmt"
	"log/slog"
	"os"
	"sync/atomic"
	"time"

	_ "github.com/ncruces/go-sqlite3/driver"
	_ "github.com/ncruces/go-sqlite3/vfs/memdb"
	"github.com/omarluq/cord"
)

var attempts atomic.Int32

func unreliable(_ context.Context, value int) (int, error) {
	attempt := attempts.Add(1)
	slog.Info("calling temporary service", "attempt", attempt)
	if attempt < 3 {
		return 0, errors.New("temporary service failure")
	}
	return value, nil
}

func confirm(_ context.Context, value int) (string, error) {
	slog.Info("retry recovered", "value", value)
	return fmt.Sprintf("recovered with %d", value), nil
}

func main() {
	slog.SetDefault(
		slog.New(
			slog.NewTextHandler(os.Stdout, nil),
		),
	)
	ctx := context.Background()
	database, err := sql.Open(
		"sqlite3",
		"file:/playground.db?vfs=memdb&_pragma=foreign_keys(1)",
	)
	if err != nil {
		panic(err)
	}
	database.SetMaxOpenConns(1)
	runtime, err := cord.New(ctx, database, cord.Options{
		PollInterval:   25 * time.Millisecond,
		MaxAttempts:    3,
		RetryBaseDelay: 100 * time.Millisecond,
		RetryMaxDelay:  200 * time.Millisecond,
	})
	if err != nil {
		panic(err)
	}
	defer func() {
		if err := errors.Join(runtime.Close(), database.Close()); err != nil {
			panic(err)
		}
	}()

	workflow := runtime.
		From("retry", unreliable).
		Then(confirm)
	result, err := workflow.Run(ctx, 7)
	if err != nil {
		panic(err)
	}
	fmt.Println(result)
}
