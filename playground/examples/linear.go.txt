package main

import (
	"context"
	"database/sql"
	"errors"
	"fmt"
	"log/slog"
	"os"
	"time"

	_ "github.com/ncruces/go-sqlite3/driver"
	_ "github.com/ncruces/go-sqlite3/vfs/memdb"
	"github.com/omarluq/cord"
)

func increment(_ context.Context, value int) (int, error) {
	slog.Info("incrementing value", "value", value)
	time.Sleep(350 * time.Millisecond)
	return value + 1, nil
}

func double(_ context.Context, value int) (int, error) {
	slog.Info("doubling value", "value", value)
	time.Sleep(350 * time.Millisecond)
	return value * 2, nil
}

func main() {
	slog.SetDefault(
		slog.New(
			slog.NewTextHandler(os.Stdout, nil),
		),
	)
	ctx := context.Background()
	database, runtime := openRuntime(ctx)
	defer closeRuntime(runtime, database)

	workflow := runtime.
		From("linear", increment).
		Then(double)

	result, err := workflow.Run(ctx, 4)
	if err != nil {
		panic(err)
	}
	fmt.Printf("result: %d\n", result)
}

func openRuntime(ctx context.Context) (*sql.DB, *cord.Cord) {
	database, err := sql.Open(
		"sqlite3",
		"file:/playground.db?vfs=memdb&_pragma=foreign_keys(1)",
	)
	if err != nil {
		panic(err)
	}
	database.SetMaxOpenConns(1)

	runtime, err := cord.New(
		ctx,
		database,
		cord.Options{PollInterval: 25 * time.Millisecond},
	)
	if err != nil {
		panic(err)
	}
	return database, runtime
}

func closeRuntime(runtime *cord.Cord, database *sql.DB) {
	if err := errors.Join(runtime.Close(), database.Close()); err != nil {
		panic(err)
	}
}
